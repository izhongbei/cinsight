```ts
/** 64 位内容哈希：两个独立的 32 位 FNV-1a（不同 offset/prime），无 BigInt 开销。 */
export interface ContentHash {
  readonly lo: number
  readonly hi: number
}

const FNV32_PRIME = 16777619
const MIX_PRIME = 2654435761

export function hashContent64(text: string): ContentHash {
  let lo = 0x811c9dc5
  let hi = 0x9e3779b9
  for (let index = 0; index < text.length; index++) {
    const code = text.charCodeAt(index)
    lo = Math.imul(lo ^ code, FNV32_PRIME) >>> 0
    hi = Math.imul(hi ^ code, MIX_PRIME) >>> 0
  }
  return { lo, hi }
}
```

- [ ] **Step 4: 在 `src/core/scan.ts` 增加 include 与条件扫描**

```ts
export interface ScannedInclude {
  readonly path: string
  readonly line: number
  readonly angled: boolean
}

const INCLUDE_RE = /^#\s*include\s*(["<])([^">\s]+)[">]/

/** 行首预处理 include（跳过块注释/行注释内的伪指令）。 */
export function scanIncludes(text: string): ScannedInclude[] {
  const stripped = stripSources(text)
  const strippedLines = stripped.split('\n')
  const result: ScannedInclude[] = []
  for (let line = 0; line < strippedLines.length; line++) {
    const match = INCLUDE_RE.exec(strippedLines[line]!.trim())
    if (!match) continue
    result.push({ path: match[2]!, line, angled: match[1] === '<' })
  }
  return result
}

export interface ConditionTimeline {
  /** 按行升序：从 line 起 condition 生效；condition undefined = 无条件 */
  readonly changes: ReadonlyArray<{ readonly line: number; readonly condition: string | undefined }>
}

const CONDITION_DIRECTIVE_RE = /^#\s*(if|ifdef|ifndef|elif|else|endif)\b(.*)$/
const MAX_CONDITION_DEPTH = 3
const MAX_CONDITION_LENGTH = 120

/** 记录 #if/#ifdef/#ifndef 条件栈的时间线（不求值，未配对 #endif 忽略）。 */
export function scanConditions(text: string): ConditionTimeline {
  const stripped = stripSources(text)
  const lines = stripped.split('\n')
  const stack: string[] = []
  const changes: Array<{ line: number; condition: string | undefined }> = []
  const pushChange = (line: number, condition: string | undefined): void => {
    changes.push({ line, condition })
  }
  for (let line = 0; line < lines.length; line++) {
    const match = CONDITION_DIRECTIVE_RE.exec(lines[line]!.trim())
    if (!match) continue
    const directive = match[1]
    const expression = match[2]!.trim()
    if (directive === 'endif') {
      stack.pop()
      pushChange(line, formatCondition(stack))
    } else if (directive === 'else' || directive === 'elif') {
      if (stack.length > 0) stack[stack.length - 1] = `#else`
      pushChange(line, formatCondition(stack))
    } else {
      stack.push(`#${directive} ${expression}`.trim())
      pushChange(line, formatCondition(stack))
    }
  }
  return { changes }
}

function formatCondition(stack: readonly string[]): string | undefined {
  if (stack.length === 0) return undefined
  const joined = stack.slice(Math.max(0, stack.length - MAX_CONDITION_DEPTH)).join(' && ')
  return joined.length > MAX_CONDITION_LENGTH ? `${joined.slice(0, MAX_CONDITION_LENGTH - 1)}…` : joined
}

/** 返回某行生效的条件（时间线二分）。 */
export function conditionAt(timeline: ConditionTimeline, line: number): string | undefined {
  const changes = timeline.changes
  let low = 0
  let high = changes.length - 1
  let condition: string | undefined
  while (low <= high) {
    const mid = (low + high) >>> 1
    if (changes[mid]!.line <= line) {
      condition = changes[mid]!.condition
      low = mid + 1
    } else {
      high = mid - 1
    }
  }
  return condition
}
```
> 说明：`stripSources` 已存在（会移除注释/字符串内容但保留换行），所以行号可靠；`#elif/#else` 用 `#else` 占位表示"另一分支"。

const resolveInclude = (fromFile: FileRecord, includePath: string): string | undefined => {
  const dir = path.posix.dirname(fromFile.path)
  const candidates = [
    path.posix.normalize(path.posix.join(dir, includePath)),
    path.posix.normalize(includePath),

  for (const candidate of candidates) if (fileIndexByPath.has(candidate)) return candidate
  for (const root of roots) {

    if (fileIndexByPath.has(candidate)) return candidate

  return undefined
}

```ts
export interface RankContext {
  readonly sourceFile: string | undefined
  readonly roots: readonly string[]
  readonly table?: SymbolTable
}

export interface DefinitionGroup {
  readonly entries: readonly SymbolEntry[]
  readonly representative: SymbolEntry
  readonly included: boolean
  readonly conditional: boolean
  readonly copies: number
}

export function groupDefinitions(entries: readonly SymbolEntry[], context: RankContext): DefinitionGroup[] {
  const ranked = rankDefinitions(entries, context)
  const groups = new Map<string, SymbolEntry[]>()
  for (const entry of ranked) {
    const hash = context.table?.contentHashOf(entry.file)
    const key = hash ? `${hash.lo}:${hash.hi}` : `file:${entry.file}`
    const list = groups.get(key)
    if (list) list.push(entry)
    else groups.set(key, [entry])
  }
  return [...groups.values()].map((groupEntries) => {
    const representative = groupEntries[0]!
    return {
      entries: groupEntries,
      representative,
      included: isIncluded(context, representative.file),
      conditional: Boolean(representative.condition),
      copies: groupEntries.length,
    }
  })
}

function isIncluded(context: RankContext, file: string): boolean {
  if (!context.table || !context.sourceFile) return false
  return context.table.isReachable(context.sourceFile, file)
}

export function formatDefinitionLabel(group: DefinitionGroup): string {
  const parts = [group.representative.name]
  if (group.included) parts.push('included')
  if (group.copies > 1) parts.push(`${group.copies} copies`)
  if (group.conditional) parts.push(group.representative.condition ?? 'conditional')
  return parts.join('  ·  ')
}
```
